### 1. 军师属性定义 (Faction.cs)

```csharp
namespace GameObjects
{
    [DataContract]
    public class Faction : GameObject
    {
        // 私有字段
        private Person advisor = null;
        private int advisorID = -1;

        /// <summary>
        /// 军师ID - 持久化字段
        /// </summary>
        [DataMember]
        public int AdvisorID
        {
            get { return this.advisorID; }
            set { this.advisorID = value; }
        }

        /// <summary>
        /// 军师对象 - 带缓存机制
        /// </summary>
        public Person Advisor
        {
            get
            {
                // 如果缓存为空且有有效ID，则从Scenario中获取
                if (this.advisor == null && this.advisorID != -1 && 
                    Session.Current.Scenario != null && 
                    Session.Current.Scenario.Persons != null)
                {
                    this.advisor = Session.Current.Scenario.Persons.GetGameObject(this.AdvisorID) as Person;
                }
                
                // 检查军师有效性
                if (this.advisor != null && 
                    (!this.advisor.Alive || !this.advisor.Available || 
                     this.advisor.BelongedFaction != this))
                {
                    this.Advisor = null;
                }
                
                return this.advisor;
            }
            set
            {
                if (value != null)
                {
                    this.AdvisorID = value.ID;
                }
                else
                {
                    this.AdvisorID = -1;
                }
                this.advisor = value;
            }
        }

        /// <summary>
        /// 军师姓名 - 用于UI显示
        /// </summary>
        public string AdvisorName
        {
            get
            {
                return ((this.Advisor != null) ? this.Advisor.Name : "----");
            }
        }
    }
}
```

## 军师任命条件

### 1. 任命条件检查

```csharp
/// <summary>
/// 检查是否可以任命军师
/// 条件：
/// 1. 有君主且君主未被俘虏
/// 2. 有合格的候选人
/// </summary>
public bool AppointAdvisorAvail()
{
    if (this.Leader != null && this.Leader.BelongedCaptive == null)
    {
        // 玩家势力检查
        if (Session.Current.Scenario.IsPlayer(this) && 
            this.AdvisorCandicate.Count > 0)
        {
            return true;
        }

        // AI势力检查
        if (!Session.Current.Scenario.IsPlayer(this) && 
            this.AIAdvisorCandicate.Count > 0)
        {
            return true;
        }
    }
    return false;
}

/// <summary>
/// 检查是否可以罢免军师
/// 条件：
/// 1. 有君主且君主未被俘虏
/// 2. 当前有军师
/// </summary>
public bool RecallAdvisorAvail()
{
    return this.Leader != null && 
           this.Leader.BelongedCaptive == null && 
           this.AdvisorID != -1;
}
```

## 军师候选人筛选

### 1. 候选人条件

军师候选人必须满足以下所有条件：

1. **不是君主** - `p != this.Leader`
2. **不是当前军师** - `p != this.Advisor`
3. **可用状态** - `p.Available`
4. **存活** - `p.Alive`
5. **未被俘虏** - `p.BelongedCaptive == null`
6. **不在部队中** - `p.LocationTroop == null`
7. **智力≥70** - `p.Intelligence >= 70`

### 2. 玩家候选人列表

```csharp
/// <summary>
/// 军师候选人列表（玩家用）
/// 返回所有符合条件的人物，不排序
/// </summary>
public PersonList AdvisorCandicate
{
    get
    {
        PersonList result = new PersonList();
        foreach (Person p in this.Persons)
        {
            if (p != this.Leader && 
                p != this.Advisor && 
                p.Available && 
                p.Alive && 
                p.BelongedCaptive == null && 
                p.LocationTroop == null && 
                p.Intelligence >= 70)
            {
                result.Add(p);
            }
        }
        return result;
    }
}
```

### 3. AI候选人列表

```csharp
/// <summary>
/// AI军师候选人列表（按智力排序）
/// AI会优先选择智力最高的人物
/// </summary>
public PersonList AIAdvisorCandicate
{
    get
    {
        PersonList result = new PersonList();

        foreach (Person p in this.AdvisorCandicate)
        {
            result.Add(p);
        }

        // 按智力降序排序
        result.PropertyName = "Intelligence";
        result.IsNumber = true;
        result.SmallToBig = false;
        result.ReSort();
        
        return result;
    }
}
```

```csharp
/// <summary>
/// 处理军师双击菜单的鼠标左键点击
/// </summary>
private void HandleAdvisorDoubleClickMenu()
{
    if (InputManager.IsDown && this.viewMove == ViewMove.Stop)
    {
        DateTime currentTime = DateTime.Now;
        Point currentPosition = new Point(InputManager.PoX, InputManager.PoY);
        
        // 检查是否为双击
        if (IsDoubleClick(currentTime, currentPosition))
        {
            // 检查是否点击在空白地形上
            if (IsEmptyTerrainClick(currentPosition))
            {
                ShowAdvisorContextMenu();
            }
        }
        
        // 更新最后点击信息
        _lastLeftClickTime = currentTime;
        _lastLeftClickPosition = currentPosition;
    }
}

/// <summary>
/// 检查是否为空白地形点击
/// </summary>
private bool IsEmptyTerrainClick(Point screenPosition)
{
    // 1. 检查是否有建筑
    if (this.CurrentArchitecture != null) return false;

    // 2. 检查是否有部队
    if (this.CurrentTroop != null) return false;

    // 3. 检查是否有路径
    if (this.CurrentRouteway != null) return false;

    // 4. 检查是否在UI元素上
    if (IsOverUIElement(screenPosition)) return false;

    return true;
}
```

```csharp
/// <summary>
/// AI自动任命军师
/// AI会选择智力最高的候选人
/// </summary>
private void AIAppointAdvisor()
{
    if (!Session.Current.Scenario.IsPlayer(this))
    {
        if (this.AppointAdvisorAvail())
        {
            PersonList candidates = this.AIAdvisorCandicate;
            if (candidates.Count > 0)
            {
                Person bestCandidate = candidates[0]; // 已按智力排序
                
                // AI只在没有军师或找到更好的候选人时才重新任命
                if (this.Advisor == null || 
                    (bestCandidate.Intelligence > this.Advisor.Intelligence + 10))
                {
                    this.AdvisorID = bestCandidate.ID;
                    this.AppointAdvisor(bestCandidate);
                }
            }
        }
    }
}
```

```csharp
/// <summary>
/// 任命军师方法
/// </summary>
public void AppointAdvisor(Person person)
{
    // 记录到年表
    Session.Current.Scenario.YearTable.addAppointAdvisorEntry(
        Session.Current.Scenario.Date, 
        person, 
        this.Leader
    );
    
    // 清空advisor缓存并重新设置
    this.advisor = null;
    this.advisor = person;
    
    // 触发事件
    if (this.OnAppointAdvisor != null)
    {
        this.OnAppointAdvisor(this.Leader, person);
    }
}
```

```csharp
{

    }
```
```csharp
}
```

## 总结

军师系统是游戏中的重要功能，主要包括：

1. **基本设定**: 军师ID、军师对象、军师姓名
2. **任命条件**: 智力≥70、可用、存活、未被俘虏、不在部队中
3. **候选人筛选**: 玩家列表和AI列表（按智力排序）
4. **能力作用**: 计略系统、技能特技、战略决策
5. **UI系统**: 军师按钮、双击菜单、右键菜单
6. **AI逻辑**: 自动任命、决策权重、战略倾向
7. **事件系统**: 任命事件、年表记录、文本消息
8. **配置文件**: XML菜单配置、枚举定义、全局变量

军师的核心作用是提供战略建议、使用计略、影响AI决策，是势力发展的重要辅助角色。
