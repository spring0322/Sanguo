{
    private float _battleCooldown = 2.0f;  // 战斗冷却时间
    private Dictionary<int, float> _lastBattleTime = new Dictionary<int, float>();
    
    public bool CanEngageBattle(int troopId, float currentTime)

public int CalculateDamage(Unit attacker, Unit defender)
{
    // 基础伤害计算
    float baseDamage = attacker.Attack * Random.Range(0.8f, 1.2f);
    
    // 防御减免
    float defense = defender.Defense;
    float damageReduction = defense / (defense + 100);
    
    // 最终伤害
    int finalDamage = (int)(baseDamage * (1 - damageReduction));
    
    return Math.Max(1, finalDamage); // 至少造成1点伤害
}
```

public class QuadTree
{
    private Rectangle _bounds;
    private List<GameObject> _objects;
    private QuadTree[] _children;
    private const int MAX_OBJECTS = 10;
    private const int MAX_LEVELS = 5;
    
    public void Insert(GameObject obj)
    {
        if (_children[0] != null)
        {
            int index = GetIndex(obj);
            if (index != -1)
            {
                _children[index].Insert(obj);
                return;
            }
        }
        
        _objects.Add(obj);
        
        if (_objects.Count > MAX_OBJECTS && _level < MAX_LEVELS)
        {
            if (_children[0] == null)
                Split();
                
            // 重新分配对象到子节点
            for (int i = _objects.Count - 1; i >= 0; i--)
            {
                int index = GetIndex(_objects[i]);
                if (index != -1)
                {
                    _children[index].Insert(_objects[i]);
                    _objects.RemoveAt(i);
                }
            }
        }
    }
}
```

public override void Update()

    MouseState mouse = Mouse.GetState();
    bool isHovering = this.MapPosition.Contains(mouse.Position);
    
    if (isHovering) {
        int delta = mouse.ScrollWheelValue - _previousScrollValue;
        if (delta != 0) {
