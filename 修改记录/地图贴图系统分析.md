# 地图贴图系统分析

## 代码功能分析

这段代码实现了一个创新的"地图贴图模式"，将头像图片固定在地图的特定世界坐标上，类似于火焰系统的工作方式。

### 核心实现原理

```csharp
// 关键技术：使用摄像机变换矩阵
sb.Begin(SpriteSortMode.Deferred, BlendState.AlphaBlend, null, null, null, null, Session.Current.Camera.Transform);

// 世界坐标定位
Vector2 worldPos = new Vector2(
    Session.Current.Camera.Position.X + Session.Current.ViewportWidth / 2,
    Session.Current.Camera.Position.Y + Session.Current.ViewportHeight / 2
);

// 绘制到世界坐标
sb.Draw(mapTexture, worldPos, Color.White);
```

### 技术要点

1. **摄像机变换**：`Session.Current.Camera.Transform` 是关键，它让图片能够跟随地图坐标系
2. **世界坐标系**：使用绝对的地图坐标而不是屏幕坐标
3. **固定位置**：图片会"粘"在地图的特定位置，地图移动时图片位置相对不变

## 改进建议

### 1. 资源管理优化

```csharp
// 改进的资源获取
private static Texture2D _cachedPortrait = null;
private static int _lastAdvisorID = -1;

private Texture2D GetAdvisorPortrait()
{
    try
    {
        if (Session.Current.CurrentPlayerFaction?.AdvisorID != null)
        {
            int advisorID = Session.Current.CurrentPlayerFaction.AdvisorID;
            
            // 缓存机制，避免重复加载
            if (_lastAdvisorID != advisorID || _cachedPortrait == null)
            {
                var advisor = Session.Current.Scenario.Persons.GetGameObject(advisorID) as Person;
                if (advisor != null)
                {
                    _cachedPortrait = ResourceManager.GetPortrait(advisor.ID);
                    _lastAdvisorID = advisorID;
                }
            }
            
            if (_cachedPortrait != null)
                return _cachedPortrait;
        }
    }
    catch (Exception ex)
    {
        System.Diagnostics.Debug.WriteLine($"[地图贴图] 获取军师头像失败: {ex.Message}");
    }
    
    // 兜底纹理
    return CreateFallbackTexture();
}

private Texture2D CreateFallbackTexture()
{
    var texture = new Texture2D(Platform.GraphicsDevice, 64, 64);
    Color[] colors = new Color[64 * 64];
    for (int i = 0; i < colors.Length; i++)
        colors[i] = Color.Gold;
    texture.SetData(colors);
    return texture;
}
```

### 2. SpriteBatch状态管理优化

```csharp
// 更安全的SpriteBatch管理
private void DrawMapPortrait()
{
    var spriteBatch = Session.Current.SpriteBatch;
    var originalTransform = Matrix.Identity;
    bool wasDrawing = false;
    
    try
    {
        // 检查当前状态
        try
        {
            // 如果SpriteBatch正在绘制，先结束
            spriteBatch.End();
            wasDrawing = true;
        }
        catch
        {
            // SpriteBatch没有在绘制状态，这是正常的
        }
        
        // 开始地图坐标系绘制
        spriteBatch.Begin(
            SpriteSortMode.Deferred, 
            BlendState.AlphaBlend, 
            SamplerState.LinearClamp,
            DepthStencilState.None,
            RasterizerState.CullCounterClockwise,
            null, 
            Session.Current.Camera.Transform
        );
        
        // 获取纹理
        var portraitTexture = GetAdvisorPortrait();
        if (portraitTexture != null)
        {
            // 计算世界坐标位置
            Vector2 worldPosition = CalculateWorldPosition();
            
            // 绘制头像
            spriteBatch.Draw(portraitTexture, worldPosition, Color.White);
        }
        
        spriteBatch.End();
        
        // 如果之前在绘制状态，恢复绘制
        if (wasDrawing)
        {
            spriteBatch.Begin();
        }
    }
    catch (Exception ex)
    {
        System.Diagnostics.Debug.WriteLine($"[地图贴图] 绘制失败: {ex.Message}");
        
        // 确保SpriteBatch状态正确
        try { spriteBatch.End(); } catch { }
        if (wasDrawing)
        {
            try { spriteBatch.Begin(); } catch { }
        }
    }
}
```

### 3. 多种定位模式

```csharp
public enum MapPortraitMode
{
    CameraCenter,    // 摄像机中心
    FixedPosition,   // 固定地图坐标
    PlayerCapital,   // 玩家都城位置
    AdvisorLocation  // 军师所在位置
}

private Vector2 CalculateWorldPosition(MapPortraitMode mode = MapPortraitMode.CameraCenter)
{
    switch (mode)
    {
        case MapPortraitMode.CameraCenter:
            return new Vector2(
                Session.Current.Camera.Position.X + Session.Current.ViewportWidth / 2,
                Session.Current.Camera.Position.Y + Session.Current.ViewportHeight / 2
            );
            
        case MapPortraitMode.FixedPosition:
            // 固定在地图的特定位置（比如左上角）
            return new Vector2(200, 200);
            
        case MapPortraitMode.PlayerCapital:
            if (Session.Current.CurrentPlayerFaction?.Capital != null)
            {
                var capital = Session.Current.CurrentPlayerFaction.Capital;
                return new Vector2(
                    capital.Position.X * Session.Current.Scenario.ScenarioMap.TileWidth,
                    capital.Position.Y * Session.Current.Scenario.ScenarioMap.TileHeight
                );
            }
            break;
            
        case MapPortraitMode.AdvisorLocation:
            if (Session.Current.CurrentPlayerFaction?.AdvisorID != null)
            {
                var advisor = Session.Current.Scenario.Persons.GetGameObject(
                    Session.Current.CurrentPlayerFaction.AdvisorID) as Person;
                if (advisor?.LocationArchitecture != null)
                {
                    var location = advisor.LocationArchitecture;
                    return new Vector2(
                        location.Position.X * Session.Current.Scenario.ScenarioMap.TileWidth,
                        location.Position.Y * Session.Current.Scenario.ScenarioMap.TileHeight
                    );
                }
            }
            break;
    }
    
    // 默认返回摄像机中心
    return CalculateWorldPosition(MapPortraitMode.CameraCenter);
}
```

### 4. 完整的地图贴图系统

```csharp
public class MapPortraitSystem
{
    private Texture2D _cachedPortrait;
    private int _lastAdvisorID = -1;
    private MapPortraitMode _currentMode = MapPortraitMode.CameraCenter;
    private float _alpha = 1.0f;
    private Vector2 _offset = Vector2.Zero;
    
    public void Update(GameTime gameTime)
    {
        // 可以添加动画效果，比如淡入淡出
        // 或者根据游戏状态调整显示模式
    }
    
    public void Draw()
    {
        if (!ShouldDrawPortrait())
            return;
            
        var spriteBatch = Session.Current.SpriteBatch;
        bool wasDrawing = false;
        
        try
        {
            // 安全地管理SpriteBatch状态
            try { spriteBatch.End(); wasDrawing = true; } catch { }
            
            // 开始地图坐标系绘制
            spriteBatch.Begin(
                SpriteSortMode.Deferred, 
                BlendState.AlphaBlend, 
                null, null, null, null, 
                Session.Current.Camera.Transform
            );
            
            var texture = GetAdvisorPortrait();
            if (texture != null)
            {
                var position = CalculateWorldPosition(_currentMode) + _offset;
                var color = Color.White * _alpha;
                
                spriteBatch.Draw(texture, position, color);
            }
            
            spriteBatch.End();
            if (wasDrawing) spriteBatch.Begin();
        }
        catch (Exception ex)
        {
            System.Diagnostics.Debug.WriteLine($"[MapPortraitSystem] 绘制错误: {ex.Message}");
            try { spriteBatch.End(); } catch { }
            if (wasDrawing) try { spriteBatch.Begin(); } catch { }
        }
    }
    
    private bool ShouldDrawPortrait()
    {
        // 可以根据游戏状态决定是否显示
        return Session.Current.CurrentPlayerFaction?.AdvisorID != null &&
               Session.Current.Scenario != null &&
               Session.Current.Camera != null;
    }
    
    public void SetMode(MapPortraitMode mode)
    {
        _currentMode = mode;
    }
    
    public void SetAlpha(float alpha)
    {
        _alpha = MathHelper.Clamp(alpha, 0f, 1f);
    }
    
    public void SetOffset(Vector2 offset)
    {
        _offset = offset;
    }
}
```
