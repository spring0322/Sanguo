## 🎮 实际使用示例

### **完整的AI回合示例**

```csharp
public class AIBattleManager
{
    private IntegratedAISkillManager _aiManager;
    
    public void Initialize()
    {
        var loader = new IntegratedSkillDataLoader();
        loader.LoadData(LoadGameSkillData());
        _aiManager = loader.GetAIManager();
    }
    
    public void ExecuteAITurn(Unit aiUnit, List<Unit> enemies, List<Unit> allies)
    {
        // 1. 创建战斗上下文
        var context = new BattleContext
        {
            CurrentTurn = GameTurn,
            OwnHP = aiUnit.CurrentHP,
            MaxHP = aiUnit.MaxHP,
            EnemyCount = enemies.Count,
            AllyCount = allies.Count,
            TerrainType = CurrentMap.TerrainType,
            Weather = CurrentWeather,
            InCombat = true
        };
        
        // 2. 获取可用技能
        var availableSkills = aiUnit.GetAvailableSkillIds();
        
        // 3. AI选择技能
        var selectedSkill = _aiManager.SelectBestSkill(availableSkills, context);
        
        // 4. 执行行动
        if (selectedSkill != null)
        {
            // 显示AI决策信息
            ShowAIDecision(aiUnit, selectedSkill);
            
            // 执行技能
            ExecuteSkill(aiUnit, selectedSkill, GetBestTarget(enemies));
            
            // 更新技能状态
            selectedSkill.Usage.UseSkill(context.CurrentTurn);
        }
        else
        {
            // 普通攻击
            ExecuteNormalAttack(aiUnit, enemies[0]);
        }
    }
    
    private void ShowAIDecision(Unit unit, Skill skill)
    {
        Console.WriteLine($"【AI决策】{unit.Name} 选择使用 {skill.Name}");
        Console.WriteLine($"理由: {skill.AIProfile.AIDescription}");
    }
}
```
