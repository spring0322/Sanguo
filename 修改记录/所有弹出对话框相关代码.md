```csharp
internal bool IsShowing
{
    get { return this.isShowing; }
    set { SetIsShowing(Session.MainGame.mainGameScreen, value); }
}

public void SetIsShowing(Screen screen, bool value)
{
    if (this.isShowing != value && Setting.Current.GlobalVariables.DialogShowTime > 0)
    {
        this.isShowing = value;
        if (value)
        {
            // 显示对话框时的处理
            screen.IsHolding = true;
            screen.PushUndoneWork(new UndoneWorkItem(UndoneWorkKind.tupianwenzi, UndoneWorkSubKind.None));
            screen.OnMouseLeftUp += new Screen.MouseLeftUp(this.screen_OnMouseLeftUp);
            this.Close(screen); // 处理队列中的对话
        }
        else
        {
            // 关闭对话框时的处理
            screen.IsHolding = false;
            screen.PopUndoneWork();
            screen.OnMouseLeftUp -= new Screen.MouseLeftUp(this.screen_OnMouseLeftUp);
            
            // 执行关闭回调
            if (this.CloseFunction != null)
            {
                this.CloseFunction();
                this.CloseFunction = null;
            }
        }
    }
}
```

```csharp
public void TriggerAdvisorAdvice(WorldOfTheThreeKingdoms.GameManager.StrategyKind strategy, object target = null)
{
    try
    {
        System.Diagnostics.Debug.WriteLine($"[TriggerAdvisorAdvice] 开始执行，策略类型: {strategy}");
        
        // A. 获取当前玩家势力
        var faction = Session.Current.Scenario.CurrentPlayer;
        if (faction == null) 
        {
            System.Diagnostics.Debug.WriteLine("[TriggerAdvisorAdvice] 当前玩家势力为null，退出");
            return;
        }

        // B. 获取军师（如果没有军师，尝试用君主代替）
        Person advisor = faction.Advisor;
        Person speaker = advisor;

        if (speaker == null) 
        {
            speaker = faction.Leader;
            System.Diagnostics.Debug.WriteLine("[TriggerAdvisorAdvice] 没有军师，使用君主作为发言人");
        }
        if (speaker == null) 
        {
            System.Diagnostics.Debug.WriteLine("[TriggerAdvisorAdvice] 连君主都没有，退出");
            return;
        }

        System.Diagnostics.Debug.WriteLine($"[TriggerAdvisorAdvice] 发言人: {speaker.Name}");

        // C. 生成建议文本
        string adviceText = GetSimpleAdviceText(speaker, strategy);
        System.Diagnostics.Debug.WriteLine($"[TriggerAdvisorAdvice] 建议文本: {adviceText}");

        // D. 验证插件是否可用
        if (this.Plugins?.tupianwenziPlugin == null)
        {
            System.Diagnostics.Debug.WriteLine("[TriggerAdvisorAdvice] tupianwenziPlugin为null，无法显示对话框");
            ShowStrategyPersonSelection(strategy, null);
            return;
        }
        
        // E. 设置对话框关闭后的回调函数
        this.Plugins.tupianwenziPlugin.SetCloseFunction(new GameDelegates.VoidFunction(() =>
        {
            System.Diagnostics.Debug.WriteLine("[TriggerAdvisorAdvice] 对话框关闭回调执行");
            ShowStrategyPersonSelection(strategy, null);
        }));

        // F. 设置说话人的文本内容
        speaker.TextResultString = adviceText;
        System.Diagnostics.Debug.WriteLine($"[TriggerAdvisorAdvice] 设置说话人文本: {speaker.TextResultString}");

        // G. 设置对话框位置和显示
        this.Plugins.tupianwenziPlugin.SetPosition(ShowPosition.Bottom, Session.MainGame.mainGameScreen);
        
        System.Diagnostics.Debug.WriteLine("[TriggerAdvisorAdvice] 准备调用SetGameObjectBranch");
        // 问题所在：使用3参数版本，直接传递建议文本作为branchName
        this.Plugins.tupianwenziPlugin.SetGameObjectBranch(speaker, speaker, adviceText);
        
        System.Diagnostics.Debug.WriteLine("[TriggerAdvisorAdvice] 准备显示对话框");
        this.Plugins.tupianwenziPlugin.IsShowing = true;
        
        System.Diagnostics.Debug.WriteLine($"[TriggerAdvisorAdvice] 显示军师建议: {adviceText}");
    }
    catch (Exception ex)
    {
        System.Diagnostics.Debug.WriteLine($"[TriggerAdvisorAdvice] 错误: {ex.Message}");
        System.Diagnostics.Debug.WriteLine($"[TriggerAdvisorAdvice] 堆栈: {ex.StackTrace}");
        ShowStrategyPersonSelection(strategy, null);
    }
}

private string GetSimpleAdviceText(Person speaker, WorldOfTheThreeKingdoms.GameManager.StrategyKind strategy)
{
    string speakerName = speaker.Name;
    
    switch (strategy)
    {
        case WorldOfTheThreeKingdoms.GameManager.StrategyKind.Search:
            return $"{speakerName}：主公，派遣能人搜索，或可有所收获。";
        case WorldOfTheThreeKingdoms.GameManager.StrategyKind.Gossip:
            return $"{speakerName}：主公，流言惑众需谨慎行事，三人成虎不可小觑。";
        case WorldOfTheThreeKingdoms.GameManager.StrategyKind.Destroy:
            return $"{speakerName}：主公，破坏敌设施可削弱其实力，兵不厌诈。";
        case WorldOfTheThreeKingdoms.GameManager.StrategyKind.Instigate:
            return $"{speakerName}：主公，煽动民心需察言观色，民心可用但需时机。";
        case WorldOfTheThreeKingdoms.GameManager.StrategyKind.Assassinate:
            return $"{speakerName}：主公，暗杀风险极大需三思，此计凶险成败难料。";
        case WorldOfTheThreeKingdoms.GameManager.StrategyKind.JailBreak:
            return $"{speakerName}：主公，劫牢救人义之所在，救人一命胜造七级浮屠。";
        case WorldOfTheThreeKingdoms.GameManager.StrategyKind.Convince:
            return $"{speakerName}：主公，以德服人方为上策，招贤纳士壮我军威。";
        case WorldOfTheThreeKingdoms.GameManager.StrategyKind.Information:
            return $"{speakerName}：主公，知己知彼百战不殆，情报为先方能制胜。";
        default:
            return $"{speakerName}：主公，此策略需要仔细考虑。";
    }
}
```

## 6. 对话框系统工作原理总结

### 成功的对话框调用模式
1. **4参数版本（推荐）**：
   ```csharp
   this.Plugins.tupianwenziPlugin.SetGameObjectBranch(speaker, gameObject, TextMessageKind.XXX, "BranchName");
   ```

2. **6参数版本（完整功能）**：
   ```csharp
   this.Plugins.tupianwenziPlugin.SetGameObjectBranch(speaker, gameObject, TextMessageKind.XXX, "BranchName", "image.jpg", "sound");
   ```

3. **3参数版本（直接文本）**：
   ```csharp
   this.Plugins.tupianwenziPlugin.SetGameObjectBranch(speaker, gameObject, "直接显示的文本内容");
   ```

### 关键步骤
1. 设置对话框位置：`SetPosition(ShowPosition.Bottom, screen)`
2. 设置关闭回调（可选）：`SetCloseFunction(callback)`
3. 设置确认对话框（可选）：`SetConfirmationDialog(...)`
4. 调用SetGameObjectBranch设置内容
5. 显示对话框：`IsShowing = true`

## 7. 问题分析

### 当前军师对话系统问题
1. **使用了3参数版本**：虽然理论上应该工作，但可能存在兼容性问题
2. **缺少TextMessageKind枚举**：没有使用标准的枚举值，可能导致XML配置查找失败
3. **可能的XML配置缺失**：如果系统查找不到对应的分支配置，可能无法正确显示

### 建议的修复方案
1. **使用4参数版本**：参考工作的示例，使用TextMessageKind枚举
2. **添加AdvisorAdvice枚举值**：在TextMessageKind中添加对应的枚举
3. **或者使用6参数版本**：直接传递文本内容，绕过XML配置查找

### 绘制区域错误问题
从CacheManager.cs可以看到已经添加了详细的错误日志和零尺寸检查，但错误仍然出现在(1091, 472)等坐标，说明需要进一步追踪调用堆栈来定位具体的UI组件。
