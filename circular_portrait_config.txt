# 圆形头像配置文件

# 是否启用圆形头像功能
# true=启用圆形头像, false=使用方形头像
EnableCircularPortraits=true

# 圆形头像应用范围
# Small=只对小尺寸头像（部队）应用圆形
# All=对所有头像应用圆形
# None=不应用圆形效果
CircularPortraitScope=Small

# 圆形边缘平滑度
# 0=硬边缘, 1-5=软边缘程度
EdgeSmoothness=2

# 是否缓存圆形头像
# true=缓存圆形版本以提高性能
# false=每次动态生成（节省内存）
CacheCircularPortraits=true

# 配置说明：
# - EnableCircularPortraits: 主开关，控制是否启用圆形头像功能
# - CircularPortraitScope: 控制哪些头像使用圆形效果
#   * Small: 推荐设置，只对部队上的小头像使用圆形，更适合UI
#   * All: 所有头像都使用圆形，包括人物详情页面
#   * None: 完全禁用圆形效果，回到原来的方形头像
# - EdgeSmoothness: 圆形边缘的平滑度，数值越高边缘越柔和
# - CacheCircularPortraits: 是否缓存圆形头像，启用可提高性能但占用更多内存

# 使用建议：
# - 对于低配置设备，建议设置 CacheCircularPortraits=false 以节省内存
# - 对于高配置设备，建议保持默认设置以获得最佳性能
# - 如果不喜欢圆形效果，可以设置 EnableCircularPortraits=false
